# 插件系统实现

## 8.1 插件系统架构

### 8.1.1 核心组件

```
插件系统
├── Plugin（插件基类）
├── PluginManager（插件管理器）
├── Container（依赖注入容器）
├── EventDispatcher（事件分发器）
└── PluginLoader（插件加载器）
```

### 8.1.2 工作流程

```
1. 加载插件 → 2. 检查依赖 → 3. 注册服务 → 4. 启动插件 → 5. 监听事件
     ↓              ↓              ↓              ↓              ↓
PluginLoader → PluginManager → Container → Plugin::boot() → EventDispatcher
```

## 8.2 核心代码实现

### 8.2.1 依赖注入容器

```php
<?php
// core/Container.php

namespace Core;

class Container
{
    private array $bindings = [];
    private array $instances = [];

    /**
     * 绑定服务
     */
    public function bind(string $abstract, callable $concrete): void
    {
        $this->bindings[$abstract] = [
            'concrete' => $concrete,
            'shared' => false,
        ];
    }

    /**
     * 绑定单例服务
     */
    public function singleton(string $abstract, callable $concrete = null): void
    {
        $this->bindings[$abstract] = [
            'concrete' => $concrete ?? function($container) use ($abstract) {
                return new $abstract();
            },
            'shared' => true,
        ];
    }

    /**
     * 解析服务
     */
    public function make(string $abstract): mixed
    {
        // 如果已经有实例，直接返回
        if (isset($this->instances[$abstract])) {
            return $this->instances[$abstract];
        }

        // 如果没有绑定，尝试自动实例化
        if (!isset($this->bindings[$abstract])) {
            return $this->resolve($abstract);
        }

        $binding = $this->bindings[$abstract];
        $instance = $binding['concrete']($this);

        // 如果是单例，保存实例
        if ($binding['shared']) {
            $this->instances[$abstract] = $instance;
        }

        return $instance;
    }

    /**
     * 别名：get
     */
    public function get(string $abstract): mixed
    {
        return $this->make($abstract);
    }

    /**
     * 自动解析类（支持构造函数依赖注入）
     */
    private function resolve(string $class): object
    {
        $reflector = new \ReflectionClass($class);

        if (!$reflector->isInstantiable()) {
            throw new \Exception("类 {$class} 不可实例化");
        }

        $constructor = $reflector->getConstructor();

        if ($constructor === null) {
            return new $class();
        }

        $parameters = $constructor->getParameters();
        $dependencies = [];

        foreach ($parameters as $parameter) {
            $type = $parameter->getType();

            if ($type === null) {
                throw new \Exception("无法解析参数 {$parameter->getName()}");
            }

            $dependencies[] = $this->make($type->getName());
        }

        return $reflector->newInstanceArgs($dependencies);
    }

    /**
     * 检查服务是否已绑定
     */
    public function has(string $abstract): bool
    {
        return isset($this->bindings[$abstract]) || isset($this->instances[$abstract]);
    }
}
```

### 8.2.2 事件分发器

```php
<?php
// core/EventDispatcher.php

namespace Core;

class EventDispatcher
{
    private array $listeners = [];
    private array $wildcards = [];

    /**
     * 注册事件监听器
     */
    public function listen(string $event, callable $listener, int $priority = 0): void
    {
        $this->listeners[$event][$priority][] = $listener;
    }

    /**
     * 注册通配符监听器
     */
    public function listenWildcard(string $pattern, callable $listener): void
    {
        $this->wildcards[$pattern][] = $listener;
    }

    /**
     * 触发事件
     */
    public function dispatch(string $event, mixed $payload = null): void
    {
        // 触发精确匹配的监听器
        if (isset($this->listeners[$event])) {
            // 按优先级排序
            krsort($this->listeners[$event]);

            foreach ($this->listeners[$event] as $priorityListeners) {
                foreach ($priorityListeners as $listener) {
                    $listener($payload);
                }
            }
        }

        // 触发通配符监听器
        foreach ($this->wildcards as $pattern => $listeners) {
            if ($this->matchWildcard($pattern, $event)) {
                foreach ($listeners as $listener) {
                    $listener($payload);
                }
            }
        }
    }

    /**
     * 触发事件（对象形式）
     */
    public function fire(object $event): void
    {
        $this->dispatch(get_class($event), $event);
    }

    /**
     * 移除事件监听器
     */
    public function forget(string $event): void
    {
        unset($this->listeners[$event]);
    }

    /**
     * 通配符匹配
     */
    private function matchWildcard(string $pattern, string $event): bool
    {
        $pattern = str_replace('*', '.*', $pattern);
        return preg_match('/^' . $pattern . '$/', $event) === 1;
    }

    /**
     * 获取所有监听器
     */
    public function getListeners(string $event = null): array
    {
        if ($event === null) {
            return $this->listeners;
        }

        return $this->listeners[$event] ?? [];
    }
}
```

```php
     */
    public function install(): bool
    {
        return true;
    }

    /**
     * 插件卸载
     */
    public function uninstall(): bool
    {
        return true;
    }

    /**
     * 插件启用
     */
    public function enable(): bool
    {
        return true;
    }

    /**
     * 插件禁用
     */
    public function disable(): bool
    {
        return true;
    }

     * 辅助方法：监听事件

    {
        $this->events->listen($event, $listener, $priority);

    protected function fire(object $event): void

    }

    /**
     * 辅助方法：获取服务

    protected function make(string $abstract): mixed
    {
```

### 8.2.5 插件管理器

```php
<?php
// core/PluginManager.php

namespace Core;

class PluginManager
{
    private Container $container;
    private EventDispatcher $events;
    private array $plugins = [];
    private array $booted = [];
    private array $enabled = [];

    public function __construct(Container $container, EventDispatcher $events)
    {
        $this->container = $container;
        $this->events = $events;
        $this->loadEnabledPlugins();
    }

    /**
     * 注册插件
     */
    public function register(Plugin $plugin): void
    {
        $name = $plugin->getName();

        // 检查是否已注册
        if (isset($this->plugins[$name])) {
            throw new \Exception("插件 {$name} 已经注册");
        }

        // 检查依赖
        $this->checkDependencies($plugin);

        // 注册插件
        $this->plugins[$name] = $plugin;

        // 调用插件的 register 方法
        $plugin->register();

        // 触发插件注册事件
        $this->events->fire(new PluginRegisteredEvent($plugin));
    }

    /**
     * 启动所有插件
     */
    public function boot(): void
    {
        foreach ($this->plugins as $name => $plugin) {
            if (!in_array($name, $this->booted) && $this->isEnabled($name)) {
                $plugin->boot();
                $this->booted[] = $name;

                // 触发插件启动事件
                $this->events->fire(new PluginBootedEvent($plugin));
            }
        }
    }

    /**
     * 检查插件依赖
     */
    private function checkDependencies(Plugin $plugin): void
    {
        $dependencies = $plugin->getDependencies();

        foreach ($dependencies as $dependency) {
            if (!isset($this->plugins[$dependency])) {
                throw new \Exception(
                    "插件 {$plugin->getName()} 依赖 {$dependency}，但该插件未注册"
                );
            }
        }
    }

    /**
     * 获取插件
     */
    public function get(string $name): ?Plugin
    {
        return $this->plugins[$name] ?? null;
    }

    /**
     * 获取所有插件
     */
    public function all(): array
    {
        return $this->plugins;
    }

    /**
     * 检查插件是否已注册
     */
    public function has(string $name): bool
    {
        return isset($this->plugins[$name]);
    }

    /**
     * 启用插件
     */
    public function enable(string $name): bool
    {
        if (!$this->has($name)) {
            throw new \Exception("插件 {$name} 不存在");
        }

        $plugin = $this->get($name);

        if ($plugin->enable()) {
            $this->enabled[] = $name;
            $this->saveEnabledPlugins();

            // 如果插件未启动，立即启动
            if (!in_array($name, $this->booted)) {
                $plugin->boot();
                $this->booted[] = $name;
            }

            return true;
        }

        return false;
    }

    /**
     * 禁用插件
     */
    public function disable(string $name): bool
    {
        if (!$this->has($name)) {
            throw new \Exception("插件 {$name} 不存在");
        }

        $plugin = $this->get($name);

        if ($plugin->disable()) {
            $this->enabled = array_diff($this->enabled, [$name]);
            $this->saveEnabledPlugins();
            return true;
        }

        return false;
    }

    /**
     * 检查插件是否已启用
     */
    public function isEnabled(string $name): bool
    {
        return in_array($name, $this->enabled);
    }

    /**
     * 安装插件
     */
    public function install(string $name): bool
    {
        if (!$this->has($name)) {
            throw new \Exception("插件 {$name} 不存在");
        }

        return $this->get($name)->install();
    }

    /**
     * 卸载插件
     */
    public function uninstall(string $name): bool
    {
        if (!$this->has($name)) {
            throw new \Exception("插件 {$name} 不存在");
        }

        // 先禁用
        $this->disable($name);

        // 卸载
        return $this->get($name)->uninstall();
    }

    /**
     * 加载已启用的插件列表
     */
    private function loadEnabledPlugins(): void
    {
        $file = APP_PATH . 'storage/plugins.json';

        if (file_exists($file)) {
            $this->enabled = json_decode(file_get_contents($file), true) ?? [];
        }
    }

    /**
     * 保存已启用的插件列表
     */
    private function saveEnabledPlugins(): void
    {
        $file = APP_PATH . 'storage/plugins.json';
        file_put_contents($file, json_encode($this->enabled, JSON_PRETTY_PRINT));
    }
}
```

### 8.2.6 插件加载器

```php
<?php
// core/PluginLoader.php

namespace Core;

class PluginLoader
{
    private PluginManager $manager;
    private string $pluginPath;

    public function __construct(PluginManager $manager, string $pluginPath)
    {
        $this->manager = $manager;
        $this->pluginPath = $pluginPath;
    }

    /**
     * 自动加载所有插件
     */
    public function loadAll(): void
    {
        $directories = glob($this->pluginPath . '/*', GLOB_ONLYDIR);

        foreach ($directories as $dir) {
            $this->loadPlugin(basename($dir));
        }
    }

    /**
     * 加载单个插件
     */
    public function loadPlugin(string $name): void
    {
        $pluginFile = $this->pluginPath . '/' . $name . '/' . $name . 'Plugin.php';

        if (!file_exists($pluginFile)) {
            throw new \Exception("插件文件不存在: {$pluginFile}");
        }

        require_once $pluginFile;

        $className = 'Plugins\\' . $name . '\\' . $name . 'Plugin';

        if (!class_exists($className)) {
            throw new \Exception("插件类不存在: {$className}");
        }

        $plugin = new $className(
            $this->manager->getContainer(),
            $this->manager->getEvents()
        );

        $this->manager->register($plugin);
    }

    /**
     * 扫描可用插件
     */
    public function scan(): array
    {
        $plugins = [];
        $directories = glob($this->pluginPath . '/*', GLOB_ONLYDIR);

        foreach ($directories as $dir) {
            $name = basename($dir);
            $infoFile = $dir . '/plugin.json';

            if (file_exists($infoFile)) {
                $plugins[$name] = json_decode(file_get_contents($infoFile), true);
            }
        }

        return $plugins;
    }
}
```
