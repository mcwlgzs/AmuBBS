# CDN 与分布式部署

## 9.1 CDN 部署方案

### 9.1.1 静态资源 CDN 化

**支持的静态资源：**
- CSS 文件
- JavaScript 文件
- 图片（用户上传的图片、头像、附件）
- 字体文件
- 其他静态文件

**CDN 配置：**

```php
<?php
// config/cdn.php

return [
    'enabled' => true,
    'provider' => 'aliyun', // aliyun, qcloud, cloudflare

    'aliyun' => [
        'domain' => 'https://cdn.amubbs.com',
        'access_key' => env('ALIYUN_CDN_KEY'),
        'secret_key' => env('ALIYUN_CDN_SECRET'),
    ],

    'qcloud' => [
        'domain' => 'https://cdn.amubbs.com',
        'secret_id' => env('QCLOUD_CDN_ID'),
        'secret_key' => env('QCLOUD_CDN_KEY'),
    ],

    'cloudflare' => [
        'domain' => 'https://cdn.amubbs.com',
        'zone_id' => env('CLOUDFLARE_ZONE_ID'),
        'api_token' => env('CLOUDFLARE_API_TOKEN'),
    ],

    // 静态资源路径
    'paths' => [
        'css' => '/assets/css',
        'js' => '/assets/js',
        'images' => '/assets/images',
        'uploads' => '/uploads',
    ],
];
```

**CDN 辅助函数：**

```php
<?php
// app/Helpers/cdn.php

/**
 * 获取 CDN URL
 */
function cdn(string $path): string
{
    $config = require __DIR__ . '/../../config/cdn.php';

    if (!$config['enabled']) {
        return $path;
    }

    $provider = $config[$config['provider']];
    $domain = rtrim($provider['domain'], '/');
    $path = ltrim($path, '/');

    return $domain . '/' . $path;
}

/**
 * 获取资源 URL（带版本号）
 */
function asset(string $path): string
{
    static $manifest = null;

    if ($manifest === null) {
        $manifestFile = APP_PATH . 'public/assets/manifest.json';
        if (file_exists($manifestFile)) {
            $manifest = json_decode(file_get_contents($manifestFile), true);
        }
    }

    // 如果有版本映射，使用带版本的文件名
    if (isset($manifest[$path])) {
        $path = $manifest[$path];
    }

    return cdn($path);
}
```

**使用示例：**

```php
<!-- 视图文件 -->
<!DOCTYPE html>
<html>
<head>
    <link rel="stylesheet" href="<?= asset('assets/css/main.css') ?>">
    <script defer src="<?= asset('assets/js/app.js') ?>"></script>
</head>
<body>
    <img src="<?= cdn('/uploads/2026/02/image.jpg') ?>" alt="图片">
</body>
</html>
```

### 9.1.2 图片上传到 CDN

```php
<?php
// app/Services/UploadService.php

namespace App\Services;

class UploadService
{
    private array $config;

    public function __construct()
    {
        $this->config = require APP_PATH . 'config/cdn.php';
    }

    /**
     * 上传图片
     */
    public function uploadImage(array $file): array
    {
        // 验证文件
        $this->validateImage($file);

        // 生成文件名
        $filename = $this->generateFilename($file['name']);
        $path = $this->getUploadPath() . '/' . $filename;

        // 上传到本地
        $localPath = APP_PATH . 'public' . $path;
        move_uploaded_file($file['tmp_name'], $localPath);

        // 如果启用 CDN，同步到 CDN
        if ($this->config['enabled']) {
            $this->syncToCDN($localPath, $path);
        }

        return [
            'url' => cdn($path),
            'path' => $path,
            'size' => filesize($localPath),
        ];
    }

    /**
     * 同步文件到 CDN
     */
    private function syncToCDN(string $localPath, string $remotePath): void
    {
        $provider = $this->config['provider'];
```

### 9.1.3 CDN 缓存刷新

```php
<?php
// app/Services/CDNService.php

namespace App\Services;

class CDNService
{
    private array $config;

    public function __construct()
    {
        $this->config = require APP_PATH . 'config/cdn.php';
    }

    /**
     * 刷新 CDN 缓存
     */
    public function purge(array $urls): bool
    {
        $provider = $this->config['provider'];

        switch ($provider) {
            case 'aliyun':
                return $this->purgeAliyun($urls);
            case 'qcloud':
                return $this->purgeQcloud($urls);
            case 'cloudflare':
                return $this->purgeCloudflare($urls);
        }

        return false;
    }

    /**
     * 刷新阿里云 CDN
     */
    private function purgeAliyun(array $urls): bool
    {
        // 使用阿里云 CDN API
        return true;
    }

    /**
     * 刷新 Cloudflare CDN
     */
    private function purgeCloudflare(array $urls): bool
    {
        $config = $this->config['cloudflare'];

        $ch = curl_init();
        curl_setopt_array($ch, [
            CURLOPT_URL => "https://api.cloudflare.com/client/v4/zones/{$config['zone_id']}/purge_cache",
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_POST => true,
            CURLOPT_HTTPHEADER => [
                'Authorization: Bearer ' . $config['api_token'],
                'Content-Type: application/json',
            ],
            CURLOPT_POSTFIELDS => json_encode(['files' => $urls]),
        ]);

        $response = curl_exec($ch);
        curl_close($ch);

        $result = json_decode($response, true);
        return $result['success'] ?? false;
    }
}
```

### 9.2.2 负载均衡配置

**Nginx 负载均衡：**

```nginx
# /etc/nginx/nginx.conf

upstream php_backend {
    # 负载均衡算法
    least_conn;  # 最少连接数
    # ip_hash;   # IP 哈希（会话保持）
    # hash $request_uri consistent;  # URL 哈希

    # Web 节点列表
    server 192.168.1.101:80 weight=3 max_fails=3 fail_timeout=30s;
    server 192.168.1.102:80 weight=2 max_fails=3 fail_timeout=30s;
    server 192.168.1.103:80 weight=1 max_fails=3 fail_timeout=30s backup;

    # 健康检查
    keepalive 32;
}

server {
    listen 80;
    server_name amubbs.com;

    location / {
        proxy_pass http://php_backend;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # 超时设置
        proxy_connect_timeout 5s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;
    }

    # 静态资源直接返回（如果使用 CDN 可以去掉）
    location ~* \.(jpg|jpeg|png|gif|ico|css|js)$ {
        proxy_pass http://php_backend;
        proxy_cache static_cache;
        proxy_cache_valid 200 30d;
        expires 30d;
    }
}

# 缓存配置
proxy_cache_path /var/cache/nginx levels=1:2 keys_zone=static_cache:100m inactive=30d max_size=1g;
```

### 9.2.3 Session 共享方案

**方案一：Redis Session（推荐）**

```php
<?php
// config/session.php

return [
    'driver' => 'redis',
    'lifetime' => 7200, // 2 小时
    'redis' => [
        'host' => '192.168.1.201',
        'port' => 6379,
        'database' => 1,
        'prefix' => 'session:',
    ],
];
```

```php
<?php
// core/Session.php

namespace Core;

class Session
{
    private \Redis $redis;
    private string $sessionId;
    private array $config;

    public function __construct()
    {
        $this->config = require APP_PATH . 'config/session.php';
        $this->initRedis();
        $this->start();
    }

    private function initRedis(): void
    {
        $this->redis = new \Redis();
        $this->redis->connect(
            $this->config['redis']['host'],
            $this->config['redis']['port']
        );
        $this->redis->select($this->config['redis']['database']);
    }

    public function start(): void
    {
        $this->sessionId = $_COOKIE['session_id'] ?? $this->generateId();

        if (!isset($_COOKIE['session_id'])) {
            setcookie('session_id', $this->sessionId, [
                'expires' => time() + $this->config['lifetime'],
                'path' => '/',
                'httponly' => true,
                'secure' => true,
                'samesite' => 'Lax',
            ]);
        }
    }

    public function get(string $key, mixed $default = null): mixed
    {
        $data = $this->redis->get($this->getKey());
        if ($data === false) {
            return $default;
        }

        $session = unserialize($data);
        return $session[$key] ?? $default;
    }

    public function set(string $key, mixed $value): void
    {
        $data = $this->redis->get($this->getKey());
        $session = $data ? unserialize($data) : [];

        $session[$key] = $value;

        $this->redis->setex(
            $this->getKey(),
            $this->config['lifetime'],
            serialize($session)
        );
    }

    private function getKey(): string
    {
        return $this->config['redis']['prefix'] . $this->sessionId;
    }

    private function generateId(): string
    {
        return bin2hex(random_bytes(16));
    }
}
```

### 9.2.5 数据库读写分离

```php
<?php
// config/database.php

return [
    'default' => 'mysql',

    'connections' => [
        'mysql' => [
            'write' => [
                'host' => '192.168.1.201', // 主库
                'port' => 3306,
            ],
            'read' => [
                [
                    'host' => '192.168.1.202', // 从库1
                    'port' => 3306,
                ],
                [
                    'host' => '192.168.1.203', // 从库2
                    'port' => 3306,
                ],
            ],
            'database' => 'amubbs',
            'username' => 'root',
            'password' => '',
            'charset' => 'utf8mb4',
        ],
    ],
];
```

```php
<?php
// core/Database.php 扩展

namespace Core;

class Database
{
    private static ?PDO $writeConnection = null;
    private static array $readConnections = [];
    private static int $readIndex = 0;

    /**
     * 获取写连接（主库）
     */
    public static function getWriteConnection(): PDO
    {
        if (self::$writeConnection === null) {
            $config = require __DIR__ . '/../config/database.php';
            $write = $config['connections']['mysql']['write'];

            self::$writeConnection = self::createConnection($write);
        }

        return self::$writeConnection;
    }

    /**
     * 获取读连接（从库，轮询）
     */
    public static function getReadConnection(): PDO
    {
        $config = require __DIR__ . '/../config/database.php';
        $reads = $config['connections']['mysql']['read'];

        // 轮询选择从库
        $index = self::$readIndex % count($reads);
        self::$readIndex++;

        if (!isset(self::$readConnections[$index])) {
            self::$readConnections[$index] = self::createConnection($reads[$index]);
        }

        return self::$readConnections[$index];
    }

    /**
     * 查询操作使用从库
     */
    public static function fetchOne(string $sql, array $params = []): ?array
    {
        $pdo = self::getReadConnection();
        $stmt = $pdo->prepare($sql);
        $stmt->execute($params);
        $result = $stmt->fetch();

        return $result ?: null;
    }

    /**
     * 写操作使用主库
     */
    public static function execute(string $sql, array $params = []): int
    {
        $pdo = self::getWriteConnection();
        $stmt = $pdo->prepare($sql);
        $stmt->execute($params);

        return $stmt->rowCount();
    }

    private static function createConnection(array $config): PDO
    {
        $dsn = sprintf(
            'mysql:host=%s;port=%d;dbname=%s;charset=utf8mb4',
            $config['host'],
            $config['port'],
            $config['database'] ?? 'amubbs'
        );

        return new PDO($dsn, $config['username'] ?? 'root', $config['password'] ?? '', [
            PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
            PDO::ATTR_PERSISTENT => true,
        ]);
    }
}
```

```php
<?php
// core/Cache.php 扩展

namespace Core;

class Cache
{
    private static ?\RedisCluster $cluster = null;

    private static function getCluster(): \RedisCluster
    {
        if (self::$cluster === null) {
            $config = require __DIR__ . '/../config/cache.php';

            self::$cluster = new \RedisCluster(
                null,
                $config['redis-cluster']['nodes']
            );
        }

        return self::$cluster;
    }

    public static function get(string $key): mixed
    {
        return self::getCluster()->get($key);
    }

    public static function set(string $key, mixed $value, int $ttl = 3600): bool
    {
        return self::getCluster()->setex($key, $ttl, $value);
    }
}
```

## 9.4 部署脚本

### 9.4.1 自动化部署脚本

```bash
#!/bin/bash
# deploy.sh

set -e

echo "开始部署 AMuBBS..."

# 1. 拉取代码
git pull origin main

# 2. 安装依赖
composer install --no-dev --optimize-autoloader

# 3. 清除缓存
php artisan cache:clear

# 4. 数据库迁移
php artisan migrate --force

# 5. 重启 PHP-FPM
sudo systemctl reload php8.2-fpm

# 6. 刷新 CDN 缓存
php artisan cdn:purge

echo "部署完成！"
```
