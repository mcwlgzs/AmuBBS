### 2.2.1 启动引导 (Bootstrap)

```php
<?php
// core/Bootstrap.php

namespace Core;

class Bootstrap
{
    private static ?self $instance = null;
    private array $config = [];
    private Database $db;
    private Cache $cache;
    private Router $router;

    private function __construct()
    {
        // 加载配置
        $this->loadConfig();

        // 初始化核心组件
        $this->initDatabase();
        $this->initCache();
        $this->initRouter();

        // 注册错误处理
        $this->registerErrorHandler();

        // 加载插件
        $this->loadPlugins();
    }

    public static function getInstance(): self
    {
        if (self::$instance === null) {
            self::$instance = new self();
        }
        return self::$instance;
    }

    public function run(): void
    {
        // 启动 OPcache 预热
        if (function_exists('opcache_compile_file')) {
            $this->warmupOpcache();
        }

        // 路由分发
        $this->router->dispatch();
    }

    private function loadConfig(): void
    {
        $configFiles = glob(__DIR__ . '/../config/*.php');
        foreach ($configFiles as $file) {
            $key = basename($file, '.php');
            $this->config[$key] = require $file;
        }
    }

    private function warmupOpcache(): void
    {
        // 预编译核心文件
        $coreFiles = [
            __DIR__ . '/Router.php',
            __DIR__ . '/Database.php',
            __DIR__ . '/Cache.php',
        ];

        foreach ($coreFiles as $file) {
            if (file_exists($file)) {
                opcache_compile_file($file);
            }
        }
    }
}
```

### 2.2.2 路由系统 (Router)

```php
<?php
// core/Router.php

namespace Core;

class Router
{
    private array $routes = [];
    private array $middlewares = [];

    /**
     * 注册 GET 路由
     */
    public function get(string $path, callable|array $handler, array $middlewares = []): void
    {
        $this->addRoute('GET', $path, $handler, $middlewares);
    }

    /**
     * 注册 POST 路由
     */
    public function post(string $path, callable|array $handler, array $middlewares = []): void
    {
        $this->addRoute('POST', $path, $handler, $middlewares);
    }

    /**
     * 路由分发
     */
    public function dispatch(): void
    {
        $method = $_SERVER['REQUEST_METHOD'];
        $path = parse_url($_SERVER['REQUEST_URI'], PHP_URL_PATH);

        // 查找匹配的路由
        $route = $this->matchRoute($method, $path);

        if ($route === null) {
            $this->handleNotFound();
            return;
        }

        // 执行中间件
        $this->runMiddlewares($route['middlewares']);

        // 执行控制器
        $this->runController($route['handler'], $route['params']);
    }

    /**
     * 路由匹配（支持动态参数）
     */
    private function matchRoute(string $method, string $path): ?array
    {
        foreach ($this->routes[$method] ?? [] as $route) {
            $pattern = $this->convertToRegex($route['path']);
            if (preg_match($pattern, $path, $matches)) {
                array_shift($matches); // 移除完整匹配
                return [
                    'handler' => $route['handler'],
                    'middlewares' => $route['middlewares'],
                    'params' => $matches,
                ];
            }
        }
        return null;
    }

    /**
     * 将路由路径转换为正则表达式
     * /thread/{id} => /^\/thread\/(\d+)$/
     */
    private function convertToRegex(string $path): string
    {
        $pattern = preg_replace('/\{(\w+)\}/', '(\d+)', $path);
        return '#^' . $pattern . '$#';
    }
}
```

```php
<?php
// core/Database.php

namespace Core;

use PDO;
use PDOException;

class Database
{
    private static ?PDO $connection = null;
    private static array $queryLog = [];

    /**
     * 获取数据库连接（单例模式）
     */
    public static function getConnection(): PDO
    {
        if (self::$connection === null) {
            $config = require __DIR__ . '/../config/database.php';

            $dsn = sprintf(
                'mysql:host=%s;port=%d;dbname=%s;charset=utf8mb4',
                $config['host'],
                $config['port'],
                $config['database']
            );

            try {
                self::$connection = new PDO($dsn, $config['username'], $config['password'], [
                    PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
                    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
                    PDO::ATTR_EMULATE_PREPARES => false, // 使用真正的预处理语句
                    PDO::ATTR_PERSISTENT => true, // 持久连接
                ]);
            } catch (PDOException $e) {
                throw new \RuntimeException('数据库连接失败: ' . $e->getMessage());
            }
        }

        return self::$connection;
    }

    /**
     * 查询单条记录
     */
    public static function fetchOne(string $sql, array $params = []): ?array
    {
        $startTime = microtime(true);

        $stmt = self::getConnection()->prepare($sql);
        $stmt->execute($params);
        $result = $stmt->fetch();

        self::logQuery($sql, $params, microtime(true) - $startTime);

        return $result ?: null;
    }

    /**
     * 查询多条记录
     */
    public static function fetchAll(string $sql, array $params = []): array
    {
        $startTime = microtime(true);

        $stmt = self::getConnection()->prepare($sql);
        $stmt->execute($params);
        $result = $stmt->fetchAll();

        self::logQuery($sql, $params, microtime(true) - $startTime);

        return $result;
    }

    /**
     * 执行写操作（INSERT/UPDATE/DELETE）
     */
    public static function execute(string $sql, array $params = []): int
    {
        $startTime = microtime(true);

        $stmt = self::getConnection()->prepare($sql);
        $stmt->execute($params);
        $affectedRows = $stmt->rowCount();

        self::logQuery($sql, $params, microtime(true) - $startTime);

        return $affectedRows;
    }

    /**
     * 获取最后插入的 ID
     */
    public static function lastInsertId(): int
    {
        return (int) self::getConnection()->lastInsertId();
    }

    /**
     * 事务处理
     */
    public static function transaction(callable $callback): mixed
    {
        $pdo = self::getConnection();

        try {
            $pdo->beginTransaction();
            $result = $callback();
            $pdo->commit();
            return $result;
        } catch (\Exception $e) {
            $pdo->rollBack();
            throw $e;
        }
    }

    /**
     * 记录 SQL 查询日志（开发模式）
     */
    private static function logQuery(string $sql, array $params, float $time): void
    {
        if (defined('DEBUG') && DEBUG) {
            self::$queryLog[] = [
                'sql' => $sql,
                'params' => $params,
                'time' => round($time * 1000, 2) . 'ms',
            ];
        }
    }

    /**
     * 获取查询日志
     */
    public static function getQueryLog(): array
    {
        return self::$queryLog;
    }
}
```

### 2.2.4 缓存系统 (Cache)

```php
<?php
// core/Cache.php

namespace Core;

use Redis;

class Cache
{
    private static ?Redis $redis = null;
    private static array $localCache = []; // 进程内缓存

    /**
     * 获取 Redis 连接
     */
    private static function getRedis(): Redis
    {
        if (self::$redis === null) {
            $config = require __DIR__ . '/../config/cache.php';

            self::$redis = new Redis();
            self::$redis->connect($config['host'], $config['port']);

            if (!empty($config['password'])) {
                self::$redis->auth($config['password']);
            }

            self::$redis->select($config['database']);

            // 设置序列化方式
            self::$redis->setOption(Redis::OPT_SERIALIZER, Redis::SERIALIZER_PHP);
        }

        return self::$redis;
    }

    /**
     * 获取缓存（三级缓存：进程内 → Redis → 数据库）
     */
    public static function get(string $key, callable $callback = null, int $ttl = 3600): mixed
    {
        // Level 1: 进程内缓存
        if (isset(self::$localCache[$key])) {
            return self::$localCache[$key];
        }

        // Level 2: Redis 缓存
        $redis = self::getRedis();
        $value = $redis->get($key);

        if ($value !== false) {
            self::$localCache[$key] = $value;
            return $value;
        }

        // Level 3: 回调函数（通常是数据库查询）
        if ($callback !== null) {
            $value = $callback();
            self::set($key, $value, $ttl);
            return $value;
        }

        return null;
    }

    /**
     * 设置缓存
     */
    public static function set(string $key, mixed $value, int $ttl = 3600): bool
    {
        // 同时写入进程内缓存和 Redis
        self::$localCache[$key] = $value;
        return self::getRedis()->setex($key, $ttl, $value);
    }

    /**
     * 删除缓存
     */
    public static function delete(string $key): bool
    {
        unset(self::$localCache[$key]);
        return self::getRedis()->del($key) > 0;
    }

    /**
     * 批量删除缓存（支持通配符）
     */
    public static function deletePattern(string $pattern): int
    {
        $redis = self::getRedis();
        $keys = $redis->keys($pattern);

        if (empty($keys)) {
            return 0;
        }

        // 清除进程内缓存
        foreach ($keys as $key) {
            unset(self::$localCache[$key]);
        }

        return $redis->del($keys);
    }

    /**
     * 缓存标签（用于批量清除相关缓存）
     */
    public static function tags(array $tags): CacheTag
    {
        return new CacheTag($tags);
    }
}

/**
 * 缓存标签类
 */
class CacheTag
{
    private array $tags;

    public function __construct(array $tags)
    {
        $this->tags = $tags;
    }

    public function get(string $key): mixed
    {
        return Cache::get($this->getTaggedKey($key));
    }

    public function set(string $key, mixed $value, int $ttl = 3600): bool
    {
        $taggedKey = $this->getTaggedKey($key);

        // 记录标签关系
        foreach ($this->tags as $tag) {
            Cache::getRedis()->sAdd("tag:{$tag}", $taggedKey);
        }

        return Cache::set($taggedKey, $value, $ttl);
    }

    public function flush(): void
    {
        foreach ($this->tags as $tag) {
            $keys = Cache::getRedis()->sMembers("tag:{$tag}");
            if (!empty($keys)) {
                Cache::getRedis()->del($keys);
            }
            Cache::getRedis()->del("tag:{$tag}");
        }
    }

    private function getTaggedKey(string $key): string
    {
        return implode(':', $this->tags) . ':' . $key;
    }
}
```

### 2.3.1 Controller 层（控制器）

```php
<?php
// app/Controllers/ThreadController.php

namespace App\Controllers;

use App\Services\ThreadService;

class ThreadController extends BaseController
{
    private ThreadService $threadService;

    public function __construct()
    {
        parent::__construct();
        $this->threadService = new ThreadService();
    }

    /**
     * 帖子列表
     */
    public function index(): void
    {
        $forumId = (int) ($_GET['fid'] ?? 0);
        $page = (int) ($_GET['page'] ?? 1);
        $pageSize = 20;

        // 调用 Service 层
        $result = $this->threadService->getThreadList($forumId, $page, $pageSize);

        // 返回响应
        if ($this->isAjax()) {
            $this->json($result);
        } else {
            $this->render('thread/index', $result);
        }
    }

    /**
     * 帖子详情
     */
    public function show(int $threadId): void
    {
        $userId = $this->getCurrentUserId();

        // 调用 Service 层
        $thread = $this->threadService->getThreadDetail($threadId, $userId);

        if ($thread === null) {
            $this->notFound('帖子不存在');
            return;
        }

        // 返回响应
        if ($this->isAjax()) {
            $this->json(['thread' => $thread]);
        } else {
            $this->render('thread/show', ['thread' => $thread]);
        }
    }

    /**
     * 创建帖子
     */
    public function create(): void
    {
        // 权限检查
        if (!$this->isLoggedIn()) {
            $this->unauthorized('请先登录');
            return;
        }

        // 验证输入
        $data = $this->validate([
            'forum_id' => 'required|integer',
            'title' => 'required|string|min:5|max:100',
            'content' => 'required|string|min:10',
        ]);

        // 调用 Service 层
        $threadId = $this->threadService->createThread(
            $this->getCurrentUserId(),
            $data['forum_id'],
            $data['title'],
            $data['content']
        );

        $this->json([
            'success' => true,
            'thread_id' => $threadId,
            'message' => '发帖成功',
        ]);
    }
}
```

### 2.3.2 Service 层（业务逻辑）

```php
<?php
// app/Services/ThreadService.php

namespace App\Services;

use App\Repositories\ThreadRepository;
use App\Repositories\PostRepository;
use App\Events\ThreadCreated;
use Core\Cache;

class ThreadService
{
    private ThreadRepository $threadRepo;
    private PostRepository $postRepo;

    public function __construct()
    {
        $this->threadRepo = new ThreadRepository();
        $this->postRepo = new PostRepository();
    }

    /**
     * 获取帖子列表
     */
    public function getThreadList(int $forumId, int $page, int $pageSize): array
    {
        $cacheKey = "thread:list:{$forumId}:{$page}";

        return Cache::tags(['thread', "forum:{$forumId}"])->get($cacheKey, function() use ($forumId, $page, $pageSize) {
            $offset = ($page - 1) * $pageSize;

            // 获取帖子列表
            $threads = $this->threadRepo->getByForumId($forumId, $offset, $pageSize);

            // 获取总数
            $total = $this->threadRepo->countByForumId($forumId);

            return [
                'threads' => $threads,
                'pagination' => [
                    'total' => $total,
                    'page' => $page,
                    'page_size' => $pageSize,
                    'total_pages' => ceil($total / $pageSize),
                ],
            ];
        }, 300); // 缓存 5 分钟
    }

    /**
     * 获取帖子详情
     */
    public function getThreadDetail(int $threadId, int $userId = 0): ?array
    {
        $cacheKey = "thread:detail:{$threadId}";

        $thread = Cache::get($cacheKey, function() use ($threadId) {
            return $this->threadRepo->getById($threadId);
        }, 600);

        if ($thread === null) {
            return null;
        }

        // 获取回复列表
        $posts = $this->postRepo->getByThreadId($threadId);

        // 增加浏览次数（异步）
        $this->incrementViews($threadId);

        return [
            'thread' => $thread,
            'posts' => $posts,
        ];
    }

    /**
     * 创建帖子
     */
    public function createThread(int $userId, int $forumId, string $title, string $content): int
    {
        // 防灌水检查
        $this->checkFloodControl($userId);

        // 敏感词过滤
        $title = $this->filterSensitiveWords($title);
        $content = $this->filterSensitiveWords($content);

        // 创建帖子
        $threadId = $this->threadRepo->create([
            'forum_id' => $forumId,
            'user_id' => $userId,
            'title' => $title,
            'created_at' => time(),
        ]);

        // 创建首帖
        $this->postRepo->create([
            'thread_id' => $threadId,
            'user_id' => $userId,
            'content' => $content,
            'created_at' => time(),
        ]);

        // 清除相关缓存
        Cache::tags(["forum:{$forumId}"])->flush();

        // 触发事件
        event(new ThreadCreated($threadId, $userId));

        return $threadId;
    }

    /**
     * 防灌水检查
     */
    private function checkFloodControl(int $userId): void
    {
        $key = "flood:user:{$userId}";
        $lastPostTime = Cache::get($key);

        if ($lastPostTime !== null && time() - $lastPostTime < 30) {
            throw new \RuntimeException('发帖太频繁，请稍后再试');
        }

        Cache::set($key, time(), 60);
    }

    /**
     * 敏感词过滤
     */
    private function filterSensitiveWords(string $text): string
    {
        // 从缓存获取敏感词列表
        $words = Cache::get('sensitive:words', function() {
            // 从数据库或配置文件加载
            return ['敏感词1', '敏感词2'];
        }, 3600);

        foreach ($words as $word) {
            $text = str_replace($word, str_repeat('*', mb_strlen($word)), $text);
        }

        return $text;
    }

    /**
     * 增加浏览次数（异步）
     */
    private function incrementViews(int $threadId): void
    {
        // 使用 Redis 计数器，定期同步到数据库
        Cache::getRedis()->incr("thread:views:{$threadId}");
    }
}
```

### 2.3.3 Repository 层（数据访问）

```php
<?php
// app/Repositories/ThreadRepository.php

namespace App\Repositories;

use Core\Database;

class ThreadRepository
{
    private string $table = 'threads';

    /**
     * 根据 ID 获取帖子
     */
    public function getById(int $id): ?array
    {
        $sql = "SELECT * FROM {$this->table} WHERE id = ? AND deleted_at IS NULL";
        return Database::fetchOne($sql, [$id]);
    }

    /**
     * 根据板块 ID 获取帖子列表
     */
    public function getByForumId(int $forumId, int $offset, int $limit): array
    {
        $sql = "
            SELECT t.*, u.username, u.avatar
            FROM {$this->table} t
            LEFT JOIN users u ON t.user_id = u.id
            WHERE t.forum_id = ? AND t.deleted_at IS NULL
            ORDER BY t.is_top DESC, t.last_post_time DESC
            LIMIT ? OFFSET ?
        ";

        return Database::fetchAll($sql, [$forumId, $limit, $offset]);
    }

    /**
     * 统计板块帖子数
     */
    public function countByForumId(int $forumId): int
    {
        $sql = "SELECT COUNT(*) as count FROM {$this->table} WHERE forum_id = ? AND deleted_at IS NULL";
        $result = Database::fetchOne($sql, [$forumId]);
        return (int) $result['count'];
    }

    /**
     * 创建帖子
     */
    public function create(array $data): int
    {
        $sql = "
            INSERT INTO {$this->table} (forum_id, user_id, title, created_at)
            VALUES (?, ?, ?, ?)
        ";

        Database::execute($sql, [
            $data['forum_id'],
            $data['user_id'],
            $data['title'],
            $data['created_at'],
        ]);

        return Database::lastInsertId();
    }

    /**
     * 更新帖子
     */
    public function update(int $id, array $data): bool
    {
        $fields = [];
        $params = [];

        foreach ($data as $key => $value) {
            $fields[] = "{$key} = ?";
            $params[] = $value;
        }

        $params[] = $id;

        $sql = "UPDATE {$this->table} SET " . implode(', ', $fields) . " WHERE id = ?";

        return Database::execute($sql, $params) > 0;
    }

    /**
     * 软删除帖子
     */
    public function delete(int $id): bool
    {
        $sql = "UPDATE {$this->table} SET deleted_at = ? WHERE id = ?";
        return Database::execute($sql, [time(), $id]) > 0;
    }
}
```
