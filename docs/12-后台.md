### 12.2.1 服务器节点管理界面

```php
<?php
// admin/route/cluster.php

namespace Admin\Controllers;

use Core\Database;
use Core\Cache;

/**
 * 集群管理控制器
 */
class ClusterController extends BaseController
{
    /**
     * 节点列表
     */
    public function index(): void
    {
        // 获取所有节点
        $nodes = Database::fetchAll("
            SELECT * FROM cluster_nodes
            ORDER BY type, created_at DESC
        ");

        // 检查节点状态
        foreach ($nodes as &$node) {
            $node['status'] = $this->checkNodeStatus($node);
        }

        $this->render('cluster/index', [
            'nodes' => $nodes,
        ]);
    }

    /**
     * 添加节点
     */
    public function create(): void
    {
        if ($_SERVER['REQUEST_METHOD'] === 'POST') {
            $data = [
                'type' => $_POST['type'], // web, mysql, redis
                'name' => $_POST['name'],
                'host' => $_POST['host'],
                'port' => (int) $_POST['port'],
                'weight' => (int) ($_POST['weight'] ?? 1),
                'config' => json_encode($_POST['config'] ?? []),
                'status' => 1, // 1=启用 0=禁用
                'created_at' => time(),
            ];

            // 测试连接
            if (!$this->testConnection($data)) {
                $this->error('节点连接测试失败');
                return;
            }

            // 插入数据库
            Database::execute("
                INSERT INTO cluster_nodes
                (type, name, host, port, weight, config, status, created_at)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?)
            ", array_values($data));

            // 更新配置文件
            $this->updateClusterConfig();

            // 清除缓存
            Cache::delete('cluster:nodes');

            $this->success('节点添加成功', '/admin/cluster');
        } else {
            $this->render('cluster/create');
        }
    }

    /**
     * 编辑节点
     */
    public function edit(int $id): void
    {
        $node = Database::fetchOne("
            SELECT * FROM cluster_nodes WHERE id = ?
        ", [$id]);

        if (!$node) {
            $this->error('节点不存在');
            return;
        }

        if ($_SERVER['REQUEST_METHOD'] === 'POST') {
            $data = [
                'name' => $_POST['name'],
                'host' => $_POST['host'],
                'port' => (int) $_POST['port'],
                'weight' => (int) $_POST['weight'],
                'config' => json_encode($_POST['config'] ?? []),
            ];

            // 测试连接
            if (!$this->testConnection(array_merge($data, ['type' => $node['type']]))) {
                $this->error('节点连接测试失败');
                return;
            }

            // 更新数据库
            Database::execute("
                UPDATE cluster_nodes
                SET name=?, host=?, port=?, weight=?, config=?
                WHERE id=?
            ", [...array_values($data), $id]);

            // 更新配置文件
            $this->updateClusterConfig();

            // 清除缓存
            Cache::delete('cluster:nodes');

            $this->success('节点更新成功', '/admin/cluster');
        } else {
            $node['config'] = json_decode($node['config'], true);
            $this->render('cluster/edit', ['node' => $node]);
        }
    }

    /**
     * 启用/禁用节点
     */
    public function toggle(int $id): void
    {
        $node = Database::fetchOne("
            SELECT * FROM cluster_nodes WHERE id = ?
        ", [$id]);

        if (!$node) {
            $this->jsonError('节点不存在');
            return;
        }

        $newStatus = $node['status'] == 1 ? 0 : 1;

        Database::execute("
            UPDATE cluster_nodes SET status = ? WHERE id = ?
        ", [$newStatus, $id]);

        // 更新配置文件
        $this->updateClusterConfig();

        // 清除缓存
        Cache::delete('cluster:nodes');

        $this->jsonSuccess($newStatus == 1 ? '节点已启用' : '节点已禁用');
    }

    /**
     * 删除节点
     */
    public function delete(int $id): void
    {
        Database::execute("DELETE FROM cluster_nodes WHERE id = ?", [$id]);

        // 更新配置文件
        $this->updateClusterConfig();

        // 清除缓存
        Cache::delete('cluster:nodes');

        $this->success('节点删除成功', '/admin/cluster');
    }

    /**
     * 测试节点连接
     */
    private function testConnection(array $node): bool
    {
        try {
            switch ($node['type']) {
                case 'web':
                    // 测试 Web 节点
                    $ch = curl_init();
                    curl_setopt_array($ch, [
                        CURLOPT_URL => "http://{$node['host']}:{$node['port']}/health",
                        CURLOPT_RETURNTRANSFER => true,
                        CURLOPT_TIMEOUT => 5,
                    ]);
                    $response = curl_exec($ch);
                    $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
                    curl_close($ch);
                    return $httpCode === 200;

                case 'mysql':
                    // 测试 MySQL 连接
                    $config = json_decode($node['config'], true);
                    $pdo = new \PDO(
                        "mysql:host={$node['host']};port={$node['port']}",
                        $config['username'] ?? 'root',
                        $config['password'] ?? ''
                    );
                    $pdo->query('SELECT 1');
                    return true;

                case 'redis':
                    // 测试 Redis 连接
                    $redis = new \Redis();
                    $redis->connect($node['host'], $node['port'], 5);
                    $redis->ping();
                    return true;

                default:
                    return false;
            }
        } catch (\Exception $e) {
            error_log("节点连接测试失败: " . $e->getMessage());
            return false;
        }
    }

    /**
     * 检查节点状态
     */
    private function checkNodeStatus(array $node): array
    {
        $cacheKey = "cluster:node:status:{$node['id']}";

        return Cache::get($cacheKey, function() use ($node) {
            $status = [
                'online' => false,
                'response_time' => 0,
                'last_check' => time(),
            ];

            $startTime = microtime(true);

            try {
                switch ($node['type']) {
                    case 'web':
                        $ch = curl_init();
                        curl_setopt_array($ch, [
                            CURLOPT_URL => "http://{$node['host']}:{$node['port']}/health",
                            CURLOPT_RETURNTRANSFER => true,
                            CURLOPT_TIMEOUT => 5,
                        ]);
                        $response = curl_exec($ch);
                        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
                        curl_close($ch);

                        $status['online'] = $httpCode === 200;
                        $status['response_time'] = round((microtime(true) - $startTime) * 1000, 2);
                        break;

                    case 'mysql':
                    case 'redis':
                        $status['online'] = $this->testConnection($node);
                        $status['response_time'] = round((microtime(true) - $startTime) * 1000, 2);
                        break;
                }
            } catch (\Exception $e) {
                $status['online'] = false;
                $status['error'] = $e->getMessage();
            }

            return $status;
        }, 60); // 缓存 1 分钟
    }

    /**
     * 更新集群配置文件
     */
    private function updateClusterConfig(): void
    {
        // 获取所有启用的节点
        $nodes = Database::fetchAll("
            SELECT * FROM cluster_nodes
            WHERE status = 1
            ORDER BY type, weight DESC
        ");

        // 按类型分组
        $config = [
            'web' => [],
            'mysql' => [],
            'redis' => [],
        ];

        foreach ($nodes as $node) {
            $config[$node['type']][] = [
                'host' => $node['host'],
                'port' => $node['port'],
                'weight' => $node['weight'],
                'config' => json_decode($node['config'], true),
            ];
        }

        // 写入配置文件
        $configContent = "<?php\n\nreturn " . var_export($config, true) . ";\n";
        file_put_contents(APP_PATH . 'config/cluster.php', $configContent);

        // 生成 Nginx 配置
        $this->generateNginxConfig($config['web']);
    }

    /**
     * 生成 Nginx 负载均衡配置
     */
    private function generateNginxConfig(array $webNodes): void
    {
        if (empty($webNodes)) {
            return;
        }

        $upstream = "upstream web_backend {\n";
        $upstream .= "    least_conn;\n\n";

        foreach ($webNodes as $node) {
            $upstream .= sprintf(
                "    server %s:%d weight=%d max_fails=3 fail_timeout=30s;\n",
                $node['host'],
                $node['port'],
                $node['weight']
            );
        }

        $upstream .= "    keepalive 32;\n";
        $upstream .= "}\n";

        // 写入 Nginx 配置片段
        file_put_contents(
            APP_PATH . 'storage/nginx/upstream.conf',
            $upstream
        );
    }
}
```

### 12.2.2 数据库表设计

```sql
-- 集群节点表
CREATE TABLE `cluster_nodes` (
  `id` INT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '节点ID',
  `type` ENUM('web', 'mysql', 'redis') NOT NULL COMMENT '节点类型',
  `name` VARCHAR(64) NOT NULL COMMENT '节点名称',
  `host` VARCHAR(255) NOT NULL COMMENT '主机地址',
  `port` INT UNSIGNED NOT NULL COMMENT '端口',
  `weight` INT UNSIGNED NOT NULL DEFAULT 1 COMMENT '权重',
  `config` TEXT COMMENT '配置信息（JSON）',
  `status` TINYINT NOT NULL DEFAULT 1 COMMENT '状态：1启用 0禁用',
  `created_at` INT UNSIGNED NOT NULL COMMENT '创建时间',
  `updated_at` INT UNSIGNED DEFAULT 0 COMMENT '更新时间',
  PRIMARY KEY (`id`),
  KEY `idx_type_status` (`type`, `status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='集群节点表';
```

### 12.3.1 基础设置

```php
<?php
// admin/route/settings.php

class SettingsController extends BaseController
{
    /**
     * 基础设置
     */
    public function index(): void
    {
        if ($_SERVER['REQUEST_METHOD'] === 'POST') {
            $settings = [
                'site_name' => $_POST['site_name'],
                'site_url' => $_POST['site_url'],
                'site_description' => $_POST['site_description'],
                'site_keywords' => $_POST['site_keywords'],
                'logo_url' => $_POST['logo_url'],
                'icp_number' => $_POST['icp_number'],
            ];

            foreach ($settings as $key => $value) {
                Database::execute("
                    INSERT INTO settings (`key`, `value`, updated_at)
                    VALUES (?, ?, ?)
                    ON DUPLICATE KEY UPDATE `value` = ?, updated_at = ?
                ", [$key, $value, time(), $value, time()]);
            }

            // 清除缓存
            Cache::deletePattern('setting:*');

            $this->success('设置保存成功');
        } else {
            // 获取所有设置
            $settings = Database::fetchAll("SELECT * FROM settings");
            $settingsMap = array_column($settings, 'value', 'key');
```
